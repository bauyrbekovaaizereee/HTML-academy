<script>
  const months = ["январь","февраль","март","апрель","май","июнь","июль","август","сентябрь","октябрь","ноябрь","декабрь"];
  const dows = ["Пн","Вт","Ср","Чт","Пт","Сб","Вс"];
  let now = new Date(), year = now.getFullYear(), month = now.getMonth();
  const cal = document.getElementById("cal");

  function draw() {
    document.getElementById("month-title").textContent = months[month] + " " + year;
    cal.innerHTML = "";
    dows.forEach(d => cal.insertAdjacentHTML("beforeend", `<div class="dow">${d}</div>`));
    const first = (new Date(year, month, 1).getDay() + 6) % 7;
    const days = new Date(year, month + 1, 0).getDate();
    for (let i = 0; i < first; i++) cal.insertAdjacentHTML("beforeend", "<div></div>");
    for (let d = 1; d <= days; d++) {
      const el = document.createElement("div");
      el.className = "day";
      el.textContent = d;
      const wd = (first + d - 1) % 7;
      if (wd > 4) el.classList.add("weekend");
      if (d === now.getDate() && month === now.getMonth() && year === now.getFullYear()) el.classList.add("today");
      el.onclick = () => {
        document.querySelectorAll(".picked").forEach(x => x.classList.remove("picked"));
        el.classList.add("picked");
        document.getElementById("picked-info").textContent = "Выбрано: " + d + " " + months[month] + " " + year + ", " + dows[wd];
      };
      cal.appendChild(el);
    }
  }
  document.getElementById("prev").onclick = () => { month--; if (month < 0) { month = 11; year--; } draw(); };
  document.getElementById("next").onclick = () => { month++; if (month > 11) { month = 0; year++; } draw(); };
  draw();

  /* ===== Переключение страниц ===== */
  function route() {
    const h = location.hash.slice(1) || "home";
    const onSchedule = (h === "schedule" || h === "add");
    document.getElementById("page-home").classList.toggle("show", !onSchedule);
    document.getElementById("page-schedule").classList.toggle("show", onSchedule);
    document.getElementById("m-home").classList.toggle("active", !onSchedule);
    document.getElementById("m-schedule").classList.toggle("active", onSchedule);
    const target = document.getElementById(h);
    if (target && h !== "home" && h !== "schedule") target.scrollIntoView();
    else window.scrollTo(0, 0);
  }
  window.addEventListener("hashchange", route);
  route();
</script>
